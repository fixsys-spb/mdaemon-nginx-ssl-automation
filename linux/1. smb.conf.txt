# =====================================================================
# Samba Share Configuration for Let's Encrypt Certificates
# =====================================================================
# IMPORTANT: 
# 1. Before adding this share, ensure you have applied the settings 
#    from 'smb.conf.global.patch.txt' to the [global] section of smb.conf.
# 2. Append this block to the END of your /etc/samba/smb.conf file.
# =====================================================================

[certs]
   comment = Let's Encrypt Certificates for MDaemon
   
   # Update this to your actual path on the Linux host.
   # Example: /root/nginxprog/letsencrypt/live
   path = /path/to/your/nginxprog/letsencrypt/live
   
   read only = yes
   browseable = yes
   guest ok = yes
   
   # Critical for LXC/Docker setups: forces Samba to access files as 'root'
   # because Nginx Proxy Manager creates certificates with root privileges.
   force user = root
